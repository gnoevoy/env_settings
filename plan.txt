next occurence of word  -> ctrl + shift + d 
{
  "key": "ctrl+shift+d",
  "command": "editor.action.addSelectionToNextFindMatch",
  "when": "editorFocus"
}

select all occurences of selected text (ctrl + shift + L)
{
  "key": "ctrl+shift+l",
  "command": "editor.action.selectHighlights",
  "when": "editorFocus"
}

all occurences of word -> ctrl + F2
{
  "key": "ctrl+f2",
  "command": "editor.action.changeAll",
  "when": "editorTextFocus && !editorReadonly"
}

Find & Replace dialog boxes
ctrl + f -> find dialog box
ctrl + shift + h -> replace dialog box
{
  "key": "ctrl+shift+h",
  "command": "editor.action.startFindReplaceAction",
  "when": "editorFocus || editorIsOpen"
}
{
  "key": "ctrl+f",
  "command": "actions.find",
  "when": "editorFocus || editorIsOpen"
}

in dialog boxes
alt + Enter -> select all occurenes of finded values
{
  "key": "alt+enter",
  "command": "search.action.openInEditor",
  "when": "hasSearchResult && searchViewletFocus"
}

alt + C -> match case
{
  "key": "alt+c",
  "command": "toggleSearchEditorCaseSensitive",
  "when": "inSearchEditor && searchInputBoxFocus"
}

alt + W -> match whole word (exact expression)
{
  "key": "alt+w",
  "command": "toggleSearchEditorWholeWord",
  "when": "inSearchEditor && searchInputBoxFocus"
}

alt + Z -> toggle word wrap 
{
  "key": "alt+z",
  "command": "editor.action.toggleWordWrap"
}

ctrl + space -> trigger suggestion (variables, functions etc.. )
{
  "key": "ctrl+space",
  "command": "editor.action.triggerSuggest",
  "when": "editorHasCompletionItemProvider && textInputFocus && !editorReadonly && !suggestWidgetVisible"
}

ctrl + shift + space -> trigger paramets hints (for functions , classes)
{
  "key": "ctrl+shift+space",
  "command": "editor.action.triggerParameterHints",
  "when": "editorHasSignatureHelpProvider && editorTextFocus"
}

ctrl + shift + v -> toggle markdown previos (for md formats)
{
  "key": "ctrl+shift+v",
  "command": "markdown.togglePreview",
  "when": "!terminalFocus && activeCustomEditorId == 'vscode.markdown.preview.editor' || editorFocus && !notebookEditorFocused && !terminalFocus && resourceLangId =~ /^(markdown|prompt|instructions|chatagent|skill)$/"
}
